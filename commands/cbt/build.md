"""
Strategy: {strategy_name}
Generated by CBT Framework

{Brief description from DISCOVERY.md}

import pandas as pd
import numpy as np
from typing import Optional, Dict, Any
from dataclasses import dataclass

# Import generated modules

from src.features import FeatureGenerator
from src.signals import SignalGenerator, Signal
{# from src.model import Model  # if ML strategy}


@dataclass
class Position:
    entry_time: pd.Timestamp
    entry_price: float
    direction: int  # 1 = long, -1 = short
    size: float
    stop_loss: Optional[float] = None
    take_profit: Optional[float] = None


class Strategy:
    """
    {Strategy description}

    Entry: {entry conditions summary}
    Exit: {exit conditions summary}
    """

    def __init__(self, config: dict):
        self.config = config
        self.params = config.get('strategy_params', {})

        # Initialize components
        self.data_loader = DataLoader()
        self.feature_gen = FeatureGenerator(self.params)
        self.signal_gen = SignalGenerator(self.params)
        {# self.model = Model(self.params)  # if ML}

        # State
        self.features: Optional[pd.DataFrame] = None
        self.signals: Optional[pd.DataFrame] = None

    def initialize(self) -> None:
        """Load data, generate features, prepare for backtesting."""
        # Load data
        data = self.data_loader.load_all()
        aligned = self.data_loader.get_aligned(
            start=self.config['time'].get('start_date'),
            end=self.config['time'].get('end_date')
        )

        # Generate features
        self.features = self.feature_gen.generate(aligned)

        {# Load model if ML strategy
        # model_path = 'checkpoints/model.pkl'
        # if Path(model_path).exists():
        #     self.model.load(model_path)}

    def get_signal(self, idx: int) -> Signal:
        """Get trading signal for given index."""
        return self.signal_gen.generate(self.features, idx)

    def calculate_position_size(self, signal: Signal,
                                equity: float,
                                price: float) -> float:
        """Calculate position size based on config and signal."""
        sizing = self.config['sizing']

        if sizing['mode'] == 'percent':
            risk_amount = equity * (sizing['percent_per_trade'] / 100)
        elif sizing['mode'] == 'fixed':
            risk_amount = sizing['fixed_amount']
        else:  # kelly
            risk_amount = equity * signal.confidence * 0.25  # quarter kelly

        # Apply leverage
        if self.config['leverage']['enabled']:
            risk_amount *= self.config['leverage']['default']

        return risk_amount / price

    def calculate_stop_loss(self, entry_price: float,
                           direction: int) -> Optional[float]:
        """Calculate stop loss price."""
        sl_config = self.config['risk']['stop_loss']
        if not sl_config['enabled']:
            return None

        if sl_config['mode'] == 'percent':
            pct = sl_config['percent'] / 100
            if direction == 1:  # long
                return entry_price * (1 - pct)
            else:  # short
                return entry_price * (1 + pct)

        return None

    def calculate_take_profit(self, entry_price: float,
                             direction: int) -> Optional[float]:
        """Calculate take profit price."""
        tp_config = self.config['risk']['take_profit']
        if not tp_config['enabled']:
            return None

        if tp_config['mode'] == 'percent':
            pct = tp_config['percent'] / 100
            if direction == 1:  # long
                return entry_price * (1 + pct)
            else:  # short
                return entry_price * (1 - pct)

        return None

    def check_exit(self, position: Position,
                   current_price: float,
                   current_idx: int) -> Optional[str]:
        """
        Check if position should be closed.
        Returns: 'tp', 'sl', 'signal', 'trailing', or None
        """
        # Stop loss
        if position.stop_loss:
            if position.direction == 1 and current_price <= position.stop_loss:
                return 'sl'
            if position.direction == -1 and current_price >= position.stop_loss:
                return 'sl'

        # Take profit
        if position.take_profit:
            if position.direction == 1 and current_price >= position.take_profit:
                return 'tp'
            if position.direction == -1 and current_price <= position.take_profit:
                return 'tp'

        # Signal reversal
        signal = self.get_signal(current_idx)
        if signal.direction != 0 and signal.direction != position.direction:
            return 'signal'

        return None
